function simpleHash(str) {
  let h = 0x811c9dc5;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = (h * 0x01000193) >>> 0;
  }
  return h.toString(16).padStart(8,'0') + btoa(str.split('').reverse().join('')).replace(/=/g,'');
}

//  ADMIN LOGIN
// ═══════════════════════════════════════════════════════════════
function doAdminLogin() {
  const now = Date.now();
  if (ST.lockUntil > now) { startLockCountdown(); return; }

  const phone = document.getElementById('loginPhone').value.replace(/\s/g,'');
  const pass = document.getElementById('loginPass').value;
  const al = document.getElementById('loginAlert');

  if (!phone || !pass) { showAlert(al,'⚠️ Enter phone and password.','error'); return; }
  if (!ST.account) { showAlert(al,'⚠️ No account found. Please create one first.','error'); return; }

  const phoneMatch = (phone === ST.account.phone || ('+'+phone) === ST.account.phone);
  const passMatch = verifyHash(pass, ST.account.passHash);

  if (!phoneMatch || !passMatch) {
    ST.loginAttempts++;
    save();
    logThreat(`Failed admin login attempt #${ST.loginAttempts} from ${phone}`);
    const rem = 5 - ST.loginAttempts;
    if (ST.loginAttempts >= 5) {
      ST.lockUntil = Date.now() + 30 * 60 * 1000;
      ST.loginAttempts = 0;
      save();
      startLockCountdown();
      return;
    }
    showAlert(al, `❌ Invalid credentials. ${rem} attempt(s) remaining.`, 'error');
    document.querySelector('.login-card').classList.add('shake');
    setTimeout(() => document.querySelector('.login-card').classList.remove('shake'), 500);
    return;
  }

  ST.loginAttempts = 0; save();
  ST.session = { role:'admin', name:'Admin', phone: ST.account.phone };
  launchApp();
}

function startLockCountdown() {
  const banner = document.getElementById('lockoutBanner');
  banner.classList.add('show');
  if (lockCountdown) clearInterval(lockCountdown);
  lockCountdown = setInterval(() => {
    const rem = ST.lockUntil - Date.now();
    if (rem <= 0) {
      clearInterval(lockCountdown);
      banner.classList.remove('show');
      document.getElementById('lockTimer').textContent = '30:00';
      return;
    }
    const m = Math.floor(rem / 60000), s = Math.floor((rem % 60000) / 1000);
    document.getElementById('lockTimer').textContent = `${m}:${s.toString().padStart(2,'0')}`;
  }, 1000);
}

let cpFails = 0;
let cpLockUntil = 0;

function doChangePass() {
  if (ST.cpLockUntil > Date.now()) {
    const rem = Math.ceil((ST.cpLockUntil - Date.now())/60000);
    showAlert(document.getElementById('cpAlert'), `⏱️ Locked. Wait ${rem} more minute(s).`, 'error'); return;
  }
  const old = document.getElementById('cpOld').value;
  const n1 = document.getElementById('cpNew').value;
  const n2 = document.getElementById('cpNew2').value;
  const al = document.getElementById('cpAlert');

  if (!old || !n1 || !n2) { showAlert(al,'⚠️ Fill all fields.','error'); return; }
  if (!verifyHash(old, ST.account.passHash)) {
    ST.cpFails++;
    save();
    if (ST.cpFails >= 3) {
      ST.cpLockUntil = Date.now() + 30 * 60 * 1000;
      ST.cpFails = 0; save();
      closeModal('m-changepass');
      doLogout();
      setTimeout(() => toast('🔒 Too many failed attempts. App locked 30 minutes.', 'error'), 500);
      return;
    }
    showAlert(al, `❌ Wrong current password. ${3 - ST.cpFails} attempt(s) left.`, 'error'); return;
  }
  if (n1.length < 6) { showAlert(al,'⚠️ New password must be at least 6 characters.','error'); return; }
  if (n1 !== n2) { showAlert(al,'⚠️ Passwords do not match.','error'); return; }
  ST.account.passHash = simpleHash(n1);
  ST.cpFails = 0; save();
  showAlert(al,'✅ Password changed successfully!','success');
  setTimeout(() => { closeModal('m-changepass'); ['cpOld','cpNew','cpNew2'].forEach(id => document.getElementById(id).value = ''); }, 1500);
}

function addStudent() {
  const id = sanitize(document.getElementById('sId').value.trim());
  const name = sanitize(document.getElementById('sName').value.trim());
  const cls = sanitize(document.getElementById('sClass').value.trim());
  const course = document.getElementById('sCourse').value;
  const house = document.getElementById('sHouse').value;
  const parent = sanitize(document.getElementById('sParent').value.trim());
  const amtPaid = parseFloat(document.getElementById('sAmtPaid').value) || 0;
  const feeStatus = document.getElementById('sFeeStatus').value;
  const al = document.getElementById('addStuAlert');

  if (!id || !name || !cls || !parent) { showAlert(al,'⚠️ Fill all required fields.','error'); return; }

  // If amtPaid > 0, log it as a transaction
  if (amtPaid > 0) {

  toast('✅ Student added!', 'success');
  ['sId','sName','sClass','sParent','sAmtPaid'].forEach(id => document.getElementById(id).value = '');
}

function getGPS() {

  navigator.geolocation.getCurrentPosition(pos => {

    ST.gps = { lat, lng, label: `${lat}, ${lng}` };
    document.getElementById('gpsText').textContent = `Lat: ${lat} · Lng: ${lng} · ±${Math.round(pos.coords.accuracy)}m`;

function initPaystack() {
  const parent = sanitize(document.getElementById('pyParent').value.trim());
  const phone = sanitize(document.getElementById('pyPhone').value.replace(/\s/g,''));
  const email = sanitize(document.getElementById('pyEmail').value.trim());
  const student = sanitize(document.getElementById('pyStudent').value.trim());
  const cls = sanitize(document.getElementById('pyClass').value.trim());
  const amount = parseFloat(document.getElementById('pyAmount').value);
  const al = document.getElementById('payAlert');

  if (!parent || !phone || !email || !student || !amount || !selPayMethod) {
    showAlert(al, '⚠️ Fill all required fields and select a payment method.', 'error'); return;
  }
  if (isNaN(amount) || amount <= 0) { showAlert(al, '⚠️ Enter a valid amount.', 'error'); return; }
  if (!/.+@.+\..+/.test(email)) { showAlert(al, '⚠️ Enter a valid email address (required for Paystack receipt).', 'error'); return; }

  const psKey = DEFAULT_PAYSTACK_KEY;

  // Map method to Paystack channel
  const channelMap = { mtn: 'mobile_money', telecel: 'mobile_money', visa: 'card' };
  const channel = channelMap[selPayMethod];

  // Paystack amount is in kobo/pesewas (multiply by 100)
  const amountInPesewas = Math.round(amount * 100);

  let handler;
  try {
    handler = PaystackPop.setup({
      key: psKey,
      email: email,
      amount: amountInPesewas,
      currency: 'GHS',
      ref: 'PTA_' + Date.now(),
      metadata: {
        custom_fields: [
          { display_name: 'Parent Name', variable_name: 'parent_name', value: parent },
          { display_name: 'Student Name', variable_name: 'student_name', value: student },
          { display_name: 'Class', variable_name: 'class', value: cls },
          { display_name: 'GPS Location', variable_name: 'gps', value: ST.gps?.label || 'N/A' },
          { display_name: 'Payment Method', variable_name: 'method', value: selPayMethod }
        ]
      },
      channels: [channel],
      phone: phone.startsWith('+') ? phone.slice(1) : phone,
      callback: function(response) { onPaystackSuccess(response, { parent, phone, email, student, cls, amount }); },
      onClose: function() { showAlert(al, '⚠️ Payment window closed. No charge was made. Please try again.', 'warn'); }
    });
    handler.openIframe();
  } catch(err) {
    // Paystack not loaded or test mode
    showAlert(al, '⚠️ Paystack unavailable (test mode). Simulating payment...', 'warn');
    setTimeout(() => onPaystackSuccess({ reference: 'SIM_' + Date.now() }, { parent, phone, email, student, cls, amount }), 1500);
  }
}

function onPaystackSuccess(response, data) {
  const vcode = genCode();
  const now = new Date();
  const tx = makeTx({
    parentName: data.parent, phone: data.phone, email: data.email,
    studentName: data.student, cls: data.cls, amount: data.amount,
    method: selPayMethod, ref: response.reference, vcode,
    gps: ST.gps?.label || 'Not captured', status: 'success'
  });
  ST.transactions.unshift(tx);

  // Mark student as paid
  const stu = ST.students.find(s => s.name.toLowerCase() === data.student.toLowerCase());
  if (stu) { stu.feeStatus = 'paid'; stu.amtPaid = (stu.amtPaid || 0) + data.amount; }

  save(); renderTransactions(); renderStudents(); updateDashboard(); updateAnalytics();

  // Show code
  document.getElementById('vcodeVal').textContent = vcode;
  document.getElementById('vcodeTime').textContent = `💳 ${now.toLocaleString('en-GB')} · 📍 ${ST.gps?.label || 'N/A'}`;
  document.getElementById('vcodeBox').style.display = 'block';
  document.getElementById('payAlert').className = 'alert';

  toast(`💰 GH₵${data.amount} payment confirmed!`, 'success');
  setTimeout(() => toast(`📱 Verification code ${vcode} sent to ${data.phone}`, 'info'), 2000);
  setTimeout(() => toast(`📧 Receipt sent to ${data.email}`, 'info'), 4000);
}

function makeTx(d) {
  return { id: 'TX' + Date.now(), ...d, timestamp: new Date().toISOString(), timeFormatted: new Date().toLocaleString('en-GB') };
}

function genCode() {
  const c = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  return Array.from({length: 8}, () => c[Math.floor(Math.random()*c.length)]).join('');
}

  const mTx = ST.transactions.filter(t => { const d=new Date(t.timestamp); return d.getMonth()===now.getMonth()&&d.getFullYear()===now.getFullYear(); });
  const dTx = ST.transactions.filter(t => new Date(t.timestamp).toDateString()===now.toDateString());
  const mTotal = mTx.reduce((s,t)=>s+t.amount,0);
  document.getElementById('stMonth').textContent = `GH₵ ${mTotal.toFixed(2)}`;
  document.getElementById('stToday').textContent = dTx.length;
  document.getElementById('stStudents').textContent = ST.students.length;
  document.getElementById('stPending').textContent = ST.students.filter(s=>s.feeStatus!=='paid').length;
  document.getElementById('monthPfill').style.width = Math.min(100,(mTotal/5000)*100)+'%';
  const tl = document.getElementById('dashTL');
  if (!ST.transactions.length) { tl.innerHTML = '<div class="empty"><div class="empty-icon">💸</div><p>No transactions yet</p></div>'; return; }
  tl.innerHTML = ST.transactions.slice(0,6).map(tx => `
    <div class="tl-item"><div class="tl-dot"></div><div class="tl-content">
      <div style="display:flex;justify-content:space-between"><div class="tl-title">${tx.parentName} → ${tx.studentName}</div><div class="tl-amt">GH₵ ${tx.amount}</div></div>
      <div class="tl-meta">⏰ ${tx.timeFormatted} · ${tx.method.toUpperCase()} · 🔑 ${tx.vcode}</div>
    </div></div>`).join('');
}

function updateAnalytics() {
  const now = new Date();
  const ws = new Date(now); ws.setDate(now.getDate()-now.getDay());
  const d = ST.transactions.filter(t=>new Date(t.timestamp).toDateString()===now.toDateString()).reduce((s,t)=>s+t.amount,0);
  const w = ST.transactions.filter(t=>new Date(t.timestamp)>=ws).reduce((s,t)=>s+t.amount,0);
  const m = ST.transactions.filter(t=>{const dt=new Date(t.timestamp);return dt.getMonth()===now.getMonth()&&dt.getFullYear()===now.getFullYear();}).reduce((s,t)=>s+t.amount,0);
  const y = ST.transactions.filter(t=>new Date(t.timestamp).getFullYear()===now.getFullYear()).reduce((s,t)=>s+t.amount,0);
  document.getElementById('anDay').textContent=`GH₵ ${d.toFixed(2)}`;
  document.getElementById('anWeek').textContent=`GH₵ ${w.toFixed(2)}`;
  document.getElementById('anMonth').textContent=`GH₵ ${m.toFixed(2)}`;
  document.getElementById('anYear').textContent=`GH₵ ${y.toFixed(2)}`;
  document.getElementById('bmMtn').textContent=`GH₵ ${ST.transactions.filter(t=>t.method==='mtn').reduce((s,t)=>s+t.amount,0).toFixed(2)}`;
  document.getElementById('bmTel').textContent=`GH₵ ${ST.transactions.filter(t=>t.method==='telecel').reduce((s,t)=>s+t.amount,0).toFixed(2)}`;
  document.getElementById('bmVisa').textContent=`GH₵ ${ST.transactions.filter(t=>t.method==='visa').reduce((s,t)=>s+t.amount,0).toFixed(2)}`;
}

function switchChart(type, btn) {
  document.querySelectorAll('.chart-tab').forEach(t=>t.classList.remove('active'));
  btn.classList.add('active');
  renderChart(type);
}

function renderChart(type) {
  const now = new Date(); let labels=[], values=[];
  if (type==='daily') {
    for(let i=6;i>=0;i--){const d=new Date(now);d.setDate(now.getDate()-i);labels.push(d.toLocaleDateString('en-GB',{weekday:'short'}));values.push(ST.transactions.filter(t=>new Date(t.timestamp).toDateString()===d.toDateString()).reduce((s,t)=>s+t.amount,0));}
  } else if (type==='weekly') {
    for(let i=3;i>=0;i--){const ws=new Date(now);ws.setDate(now.getDate()-now.getDay()-i*7);const we=new Date(ws);we.setDate(ws.getDate()+7);labels.push(`W${4-i}`);values.push(ST.transactions.filter(t=>{const d=new Date(t.timestamp);return d>=ws&&d<we;}).reduce((s,t)=>s+t.amount,0));}
  } else {
    for(let i=5;i>=0;i--){const m=new Date(now.getFullYear(),now.getMonth()-i,1);labels.push(m.toLocaleDateString('en-GB',{month:'short'}));values.push(ST.transactions.filter(t=>{const d=new Date(t.timestamp);return d.getMonth()===m.getMonth()&&d.getFullYear()===m.getFullYear();}).reduce((s,t)=>s+t.amount,0));}
  }
  const mx = Math.max(...values,1);
  document.getElementById('chartBars').innerHTML = values.map(v=>`<div class="cbar-col"><div class="cbar" style="height:${(v/mx)*110}px" data-v="GH₵${v.toFixed(0)}"></div></div>`).join('');
  document.getElementById('chartLbls').innerHTML = labels.map(l=>`<div style="flex:1;text-align:center;font-size:.65rem;color:var(--muted)">${l}</div>`).join('');
}

// ═══════════════════════════════════════════════════════════════
//  FLUID NAVIGATION
// ═══════════════════════════════════════════════════════════════
// Tells each visible circle how far it sits from the toggle (--n) so the
// open/close animation can stagger. Runs on open, so hidden parent-mode
// items never leave gaps.
function layoutFluid() {
  const nav = document.getElementById('fluidNav');
  const items = [...document.querySelectorAll('#mainNav .nav-tab, #mainNav .fluid-action')].filter(t => t.style.display !== 'none');
  items.forEach((t, i) => t.style.setProperty('--n', i + 1));
  nav.style.setProperty('--total', items.length);
}

function toggleFluid(force) {
  const nav = document.getElementById('fluidNav');
  const btn = document.getElementById('fluidToggle');
  const open = (typeof force === 'boolean') ? force : !nav.classList.contains('open');
  if (open) layoutFluid();
  nav.classList.toggle('open', open);
  btn.setAttribute('aria-expanded', open);
  btn.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
}

// Called when a circle is tapped: switch page, scroll up, then fold the menu away
function pickTab(tab, btn) {
  switchTab(tab, btn);
  window.scrollTo({ top: 0, behavior: 'smooth' });
  setTimeout(() => toggleFluid(false), 160);
}

document.addEventListener('keydown', e => { if (e.key === 'Escape') toggleFluid(false); });

// ═══════════════════════════════════════════════════════════════
//  THEME (light / dark) -- saved in localStorage, applied via <html data-theme="...">
// ═══════════════════════════════════════════════════════════════
const THEME_KEY = 'edupay_theme';

function currentTheme() {
  return document.documentElement.getAttribute('data-theme') === 'light' ? 'light' : 'dark';
}

function applyTheme(theme, persist) {
  document.documentElement.setAttribute('data-theme', theme);
  const goingLight = theme === 'dark';               // what the button will do on the NEXT tap
  const btn = document.getElementById('themeBtn');
  const lbl = document.getElementById('themeLabel');
  if (btn) {
    btn.setAttribute('aria-label', goingLight ? 'Switch to light mode' : 'Switch to dark mode');
    btn.setAttribute('aria-pressed', String(theme === 'light'));
  }
  if (lbl) lbl.textContent = goingLight ? 'Light mode' : 'Dark mode';
  const meta = document.querySelector('meta[name="theme-color"]');
  if (meta) meta.setAttribute('content', theme === 'light' ? '#dde7f7' : '#0a1628');
  if (persist) { try { localStorage.setItem(THEME_KEY, theme); } catch (e) {} }
}

function toggleTheme() {
  const next = currentTheme() === 'dark' ? 'light' : 'dark';
  applyTheme(next, true);
  toast(next === 'light' ? 'Light mode on' : 'Dark mode on', 'info');
  setTimeout(() => toggleFluid(false), 450);         // let the change show, then fold the menu
}

// ═══════════════════════════════════════════════════════════════
//  RANDOM PTA / SCHOOL BACKGROUND (Pexels -- free to use)
//  A different photo is picked on every page load. If one fails to load,
//  the next is tried; if all fail, the navy gradient stays.
//  To add more: use any Pexels photo ID (the number at the end of its page URL).
// ═══════════════════════════════════════════════════════════════
const PTA_PHOTO_IDS = [
  5415447,  // teacher and parent meeting
  5905617,  // teacher with schoolchildren
  5905488,  // teacher explaining a lesson
  8419197,  // teacher talking with students
  8419210,  // classroom discussion
  8423116,  // students at desks
  5212350,  // teacher helping students
  7692475   // teacher guiding a school project
];

function initBackground() {
  const layer = document.getElementById('bgPhoto');
  if (!layer) return;
  const urls = PTA_PHOTO_IDS.map(id => `https://images.pexels.com/photos/${id}/pexels-photo-${id}.jpeg?auto=compress&cs=tinysrgb&w=1600`);
  const start = Math.floor(Math.random() * urls.length);
  const queue = urls.slice(start).concat(urls.slice(0, start));   // random start, then the rest as fallbacks
  (function tryNext() {
    const url = queue.shift();
    if (!url) return;
    const img = new Image();
    img.onload = () => {
      layer.style.backgroundImage = `url("${url}")`;
      requestAnimationFrame(() => layer.classList.add('ready'));
    };
    img.onerror = tryNext;
    img.src = url;
  })();
}
